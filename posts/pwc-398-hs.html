Task 1: Divisible Average

import           Control.Exception (assert, evaluate)



mean :: [Int] -> Int

mean [] = 0

mean xs = sum xs `div` length xs



solve :: [Int] -> Int

solve =  mean . filter (\\x -> even x && x `mod` 3 == 0)

Task 2: String Difference

import           Control.Exception (assert, evaluate)

import           Data.Char         (ord)

import           Data.Function     (on)

import           Data.List         (groupBy, sortOn)

import           Data.Ord          (comparing)



diff :: String -> [Int]

diff xs =

  let xs\' = map ord xs

  in map (uncurry $ flip (-)) $ zip xs\' (tail xs\')



solve :: [String] -> String

solve xs

  = head

  . map (fst . head)

  . filter ((==) 1 . length)

  . groupBy (\\(_, ys) (_, zs) -> ys == zs)

  . sortOn snd

  $ zip xs (map diff xs)
